-- 発注データ
create table orders (
    orderid               integer generated by default as identity primary key,
    orderdate             date,
    arrivalplandate       date,
    supplierid            integer not null,
    registdate            timestamp,
    modifydate            timestamp
);

-- 発注明細データ
create table orderdetails (
    orderdetailid         integer generated by default as identity primary key,
    orderid               integer not null,
    detailno              integer,
    itemid                integer not null,
    netweight             double precision,
    netweightunittypeid   integer,
    percase               integer,
    quantity              double precision,
    unittypeid            integer,
    unitprice             integer,
    price                 integer,
    registdate            timestamp,
    modifydate            timestamp,
    constraint uq_orderdetails_no unique (orderid, detailno)
);

-- 入荷データ
create table arrivals (
    arrivalid             integer generated by default as identity primary key,
    arrivaldate           date,
    supplierid            integer not null,
    orderid               integer,
    registdate            timestamp,
    modifydate            timestamp
);

-- 入荷明細データ
create table arrivaldetails (
    arrivaldetailid       integer generated by default as identity primary key,
    arrivalid             integer not null,
    detailno              integer,
    itemid                integer not null,
    netweight             double precision,
    netweightunittypeid   integer,
    percase               integer,
    quantity              double precision,
    unittypeid            integer,
    unitprice             integer,
    price                 integer,
    expirydate            date,
    pieceunittypeid       integer,
    pieceweight           double precision,
    weight                double precision,
    pricepergram          double precision,
    registdate            timestamp,
    modifydate            timestamp,
    constraint uq_arrivaldetails_no unique (arrivalid, detailno)
);

--製造計画データ
create table plans (
    planid                integer generated by default as identity primary key,
    plandate              date not null,
    itemid                integer not null,
    unittypeid            integer,
    registdate            timestamp,
    modifydate            timestamp,
    constraint uq_plans_date_item unique (plandate, itemid)
);

--製造計画明細データ
create table plandetails (
    plandetailid          integer generated by default as identity primary key,
    planid                integer not null,
    storeid               integer not null,
    quantity              double precision,
    registdate            timestamp,
    modifydate            timestamp,
    constraint uq_plandetails_store unique (planid, storeid)
);

--製造データ
create table instructions (
    instructionid         integer generated by default as identity primary key,
    planid                integer,
    itemid                integer not null,
    planquantity          double precision,
    realquantity          double precision,
    unittypeid            integer,
    startdate             date,
    finishdate            date,
    registdate            timestamp,
    modifydate            timestamp,
    constraint uq_instructions_plan unique (planid)
);

--製造明細データ
create table instructiondetails (
    instructiondetailid   integer generated by default as identity primary key,
    instructionid         integer not null,
    storeid               integer not null,
    quantity              double precision,
    registdate            timestamp,
    modifydate            timestamp,
    constraint uq_instructiondetails_store unique (instructionid, storeid)
);


--製造消費データ
create table instructionconsumptions (
    instructionconsumptionid integer generated by default as identity primary key,
    instructionid         integer not null,
    inventoryid           integer,
    itemid                integer not null,
    quantity              double precision not null,
    unittypeid            integer,
    unitprice             double precision,
    registdate            timestamp,
    modifydate            timestamp
);

--在庫データ
create table inventories (
    inventoryid           integer generated by default as identity primary key,
    itemid                integer not null,
    arrivaldate           date,
    expirydate            date,
    quantity              double precision,
    unittypeid            integer,
    weight                double precision,
    pricepergram          double precision,
    arrivaldetailid       integer,
    instructionid         integer,
    registdate            timestamp,
    modifydate            timestamp
);

--在庫履歴データ
create table inventorylogs (
    inventorylogid        integer generated by default as identity primary key,
    transactiondate       timestamp not null,
    transactionstatus     integer not null,
    itemid                integer not null,
    inventoryid           integer,
    quantity              double precision,
    unittypeid            integer,
    afterquantity         double precision,
    unitprice             double precision,
    sourceid              integer,
    operatorid            integer,
    registdate            timestamp,
    constraint ck_inventorylogs_status check (transactionstatus between 1 and 8)
);
