-- 品目区分マスタ
create table itemtypes (
    itemtypeid            integer generated by default as identity primary key,
    itemtypename          varchar(255) not null,
    orderno               integer,
    productflag           integer default 0,
    recipeneedflag        integer default 0,
    lossrateneedflag      integer default 0,
    pieceneedflag         integer default 0,
    sellingpriceneedflag  integer default 0,
    mainsupplierneedflag  integer default 0,
    expirydateneedflag    integer default 0,
    weightneedflag        integer default 0,
    formbackcolor         varchar(255),
    buttonforecolor       varchar(255),
    buttonbackcolor       varchar(255),
    registdate            timestamp,
    modifydate            timestamp
);

-- 単位マスタ
create table unittypes (
    unittypeid            integer generated by default as identity primary key,
    unittypename          varchar(255) not null,
    orderno               integer,
    unitcategory          integer,
    basefactor            integer,
    nonconvertibleflag    integer default 0,
    unusedflag            integer default 0,
    registdate            timestamp,
    modifydate            timestamp,
    constraint ck_unittypes_category check (unitcategory in (1, 2, 3))
);

-- 店舗マスタ
create table stores (
    storeid               integer generated by default as identity primary key,
    storecode             varchar(10),
    storename             varchar(50) not null,
    storeshortname        varchar(255),
    centerflag            integer default 0,
    orderno               integer,
    unusedflag            integer default 0,
    registdate            timestamp,
    modifydate            timestamp
);

-- 仕入先マスタ
create table suppliers (
    supplierid            integer generated by default as identity primary key,
    suppliercode          varchar(12),
    suppliername          varchar(50) not null,
    postno                varchar(255),
    address1              varchar(255),
    address2              varchar(255),
    address3              varchar(255),
    telno                 varchar(255),
    faxno                 varchar(255),
    itemtypeid            integer,
    unusedflag            integer default 0,
    registdate            timestamp,
    modifydate            timestamp,
    constraint uq_suppliers_code unique (suppliercode)
);

-- 調整理由マスタ
create table ajusttypes (
    ajusttypeid           integer generated by default as identity primary key,
    ajusttypename         varchar(255) not null,
    orderno               integer,
    unusedflag            integer default 0,
    registdate            timestamp,
    modifydate            timestamp
);

-- 担当者マスタ
create table operators (
    operatorid            integer generated by default as identity primary key,
    operatorcode          varchar(12) not null,
    operatorname          varchar(40) not null,
    passwordhash          varchar(64) not null,
    passwordsalt          varchar(32) not null,
    roleid                integer not null default 2,
    lastlogindate         timestamp,
    orderno               integer,
    unusedflag            integer default 0,
    registdate            timestamp,
    modifydate            timestamp,
    constraint uq_operators_code unique (operatorcode),
    constraint ck_operators_role check (roleid in (1, 2))
);

--品目マスタ
create table items (
    itemid                integer generated by default as identity primary key,
    itemcode              varchar(255),
    itemname              varchar(255) not null,
    jancode               varchar(255),
    itemtypeid            integer not null,
    unittypeid            integer not null,
    lossrate              double precision default 1,
    shipment              integer,
    pieceunittypeid       integer,
    pieceweight           double precision,
    originalsellingprice  integer,
    instoresellingprice   integer,
    costprice             integer,
    mainsupplierid        integer,
    expirydateneedflag    integer default 0,
    measuretype           integer,
    unusedflag            integer default 0,
    registdate            timestamp,
    modifydate            timestamp
);

--レシピマスタ
create table recipes (
    recipeid              integer generated by default as identity primary key,
    itemid                integer not null,
    comment               text,
    registdate            timestamp,
    modifydate            timestamp
);

--レシピ明細マスタ
create table recipedetails (
    recipedetailid        integer generated by default as identity primary key,
    recipeid              integer not null,
    detailno              integer,
    itemid                integer not null,
    quantity              double precision,
    unittypeid            integer,
    comment               text,
    registdate            timestamp,
    modifydate            timestamp,
    constraint uq_recipedetails_no unique (recipeid, detailno)
);
